# ====================================================
#  OCR-Korrekturen Konfiguration
# ====================================================

# === EINFACHE ERSETZUNGEN ===
# Format: fehlerhaft=korrekt
Rechnu ng=Rechnung
Gu tschrift=Gutschrift
Liefe rschein=Lieferschein
Rech nung=Rechnung
Be leg=Beleg
Ange bot=Angebot

# === LABEL-WÖRTER (Automatische Abstandskorrektur) ===
# Diese Wörter werden automatisch von angrenzenden Werten getrennt
@labels=Auftragsnummer,Datum,Kundennummer,Belegnummer,Standort,Lieferdatum,Besteller,Seite,Rechnungsnummer,Rechnungsdatum

# === REGEX-ERSETZUNGEN ===
# Format: @regex=Muster=>Ersetzung
# Trenne lange Nummern von Datum/Jahr-Angaben
@regex=(\d{6,})(\d{4}/\d{1,2})=>$1 $2

# === EIGENE WÖRTER ===
# Werden von der Wortkorrektur nie verändert und dienen als Korrekturziel
# (z. B. Firmen- und Fachbegriffe, die oft falsch erkannt werden)
@dictionary=Rechnung,Gutschrift,Lieferschein,Bestellung,Angebot,Mahnung,Sammelrechnung,Stornorechnung,Beleg,Auftrag,Lieferung,Widmann,GmbH,KG,AG,UG,OHG,Rechnungsnummer,Belegnummer,Auftragsnummer,Kundennummer

# === WORTKORREKTUR ===
# Einzelne Wörter werden nur korrigiert, wenn der Fehler eine typische OCR-Verwechslung ist
# (rn/m, cl/d, 1/l/i, 0/o, 5/s, 8/b, q/g, a/ä, o/ö, u/ü ...) und das richtige Wort in der
# mitgelieferten Fachwortliste oder in @dictionary steht. Wörter, die die Windows-Rechtschreibprüfung
# (installierte Sprachpakete) kennt oder die häufig in den eigenen Belegen vorkommen, bleiben unverändert.
# Kein Download nötig. Ausschalten mit:
# @wortkorrektur=aus

# === HÄUFIGE OCR-FEHLER ===
# Zahlen/Buchstaben-Verwechslungen in Wörtern (z. B. "Liefersche1n") korrigiert die Wortkorrektur.
# Keine Regeln wie 0=O eintragen - sie würden jede Zahl im Text verändern.

# Häufige deutsche Begriffe
Rechnungs nummer=Rechnungsnummer
Liefer schein=Lieferschein
Auftrags nummer=Auftragsnummer
Kunden nummer=Kundennummer

# === FIRMEN UND NAMEN ===
# Fügen Sie hier häufig vorkommende Firmennamen hinzu
Widma nn=Widmann
Mül ler=Müller
Schm idt=Schmidt

# === DATUMSFORMATE ===
# OCR verwechselt manchmal Punkte und Kommas
# (wird automatisch durch Datumsparser behandelt)

# ====================================================
#  ANLEITUNG ZUM HINZUFÜGEN NEUER KORREKTUREN
# ====================================================
#
# 1. DIREKTE ERSETZUNG:
#    fehlerhaft=korrekt
#
# 2. LABEL HINZUFÜGEN:
#    @labels=existierende,neue_labels
#
# 3. REGEX PATTERN:
#    @regex=(\w+)\s+(\w+)=>$1$2
#
# 4. WÖRTERBUCH ERWEITERN:
#    @dictionary=Wort1,Wort2,Wort3
#
# 5. WORTKORREKTUR AUSSCHALTEN:
#    @wortkorrektur=aus
#
# ====================================================
